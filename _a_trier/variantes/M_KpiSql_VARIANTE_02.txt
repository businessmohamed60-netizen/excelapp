Attribute VB_Name = "M_KpiSql"
Option Explicit

' ============================================================
' M_KpiSql — KPI calculés en SQL
' ============================================================

Public Function CalculerTousKpiSql(ByVal annee As Long) As Collection
    Dim result As New Collection
    If Not DbOpen() Then
        ' Fallback XML
        Set CalculerTousKpiSql = CalculerTousKpi(annee)
        Exit Function
    End If

    ' Taux de clôture audits
    Dim rs As Object
    Set rs = DbQuery("SELECT COUNT(*) AS total, SUM(CASE WHEN Statut='Clôturé' THEN 1 ELSE 0 END) AS clos " & _
        "FROM audits WHERE strftime('%Y', DateDebut) = ?", Array(CStr(annee)))
    Dim total As Long, clos As Long, taux As Double
    If Not rs.EOF Then total = Nz(rs("total")): clos = Nz(rs("clos"))
    rs.Close
    If total > 0 Then taux = clos / total * 100
    result.Add EvaluerKpi("Taux de clôture audits", taux, 90, "Haut", "%")

    ' NC ouvertes
    Set rs = DbQuery("SELECT COUNT(*) AS n FROM nc WHERE Statut <> 'Clôturée'")
    result.Add EvaluerKpi("NC ouvertes", Nz(rs("n")), 5, "Bas", "")
    rs.Close

    ' Taux de clôture NC
    Set rs = DbQuery("SELECT COUNT(*) AS total, SUM(CASE WHEN Statut='Clôturée' THEN 1 ELSE 0 END) AS clos FROM nc")
    Dim nTotal As Long, nClos As Long
    If Not rs.EOF Then nTotal = Nz(rs("total")): nClos = Nz(rs("clos"))
    rs.Close
    result.Add EvaluerKpi("Taux de clôture NC", IIf(nTotal > 0, nClos / nTotal * 100, 0), 85, "Haut", "%")

    ' CAPA en retard
    Set rs = DbQuery("SELECT COUNT(*) AS n FROM capa WHERE Statut <> 'Clôturée' AND DateEcheance < ?", _
        Array(Format$(Date, "yyyy-mm-dd")))
    result.Add EvaluerKpi("CAPA en retard", Nz(rs("n")), 2, "Bas", "")
    rs.Close

    ' 8D ouverts
    Set rs = DbQuery("SELECT COUNT(*) AS n FROM huitd WHERE D8_Date IS NULL OR D8_Date = ''")
    result.Add EvaluerKpi("8D ouverts", Nz(rs("n")), 3, "Bas", "")
    rs.Close

    ' Étalonnages échus
    Set rs = DbQuery("SELECT COUNT(*) AS n FROM moyenmesure WHERE Statut='En service' AND ProchainEtalonnage < ?", _
        Array(Format$(Date, "yyyy-mm-dd")))
    result.Add EvaluerKpi("Étalonnages échus", Nz(rs("n")), 0, "Bas", "")
    rs.Close

    ' Actions AMDEC en retard
    Set rs = DbQuery("SELECT COUNT(*) AS n FROM ligneamdec WHERE StatutAction <> 'Clôturée' AND DateEcheanceAction < ?", _
        Array(Format$(Date, "yyyy-mm-dd")))
    result.Add EvaluerKpi("Actions AMDEC en retard", Nz(rs("n")), 0, "Bas", "")
    rs.Close

    ' Lignes AMDEC critiques
    Set rs = DbQuery("SELECT COUNT(*) AS n FROM ligneamdec WHERE Criticite='Critique' AND StatutAction <> 'Clôturée'")
    result.Add EvaluerKpi("Lignes AMDEC critiques ouvertes", Nz(rs("n")), 0, "Bas", "")
    rs.Close

    ' Revues documentaires échues
    Set rs = DbQuery("SELECT COUNT(*) AS n FROM document WHERE Statut='Approuvé' AND DateRevuePrevue < ?", _
        Array(Format$(Date, "yyyy-mm-dd")))
    result.Add EvaluerKpi("Revues documentaires échues", Nz(rs("n")), 0, "Bas", "")
    rs.Close

    ' Exigences non conformes
    Set rs = DbQuery("SELECT COUNT(*) AS n FROM exigence WHERE Applicable=1 AND Conformite IN ('Non conforme','Partielle')")
    result.Add EvaluerKpi("Exigences non conformes", Nz(rs("n")), 0, "Bas", "")
    rs.Close

    Set CalculerTousKpiSql = result
End Function

Private Function Nz(ByVal v As Variant) As Long
    If IsNull(v) Or IsEmpty(v) Then Nz = 0 Else Nz = CLng(v)
End Function
