Public Function Load(ByVal collection As String) As Object
    If g_Config.DataMode = "SQL" Then
        Set Load = LoadSql(collection)
    Else
        Set Load = LoadXml(collection)
    End If
End Function

Private Function LoadSql(ByVal collection As String) As Object
    Dim result As Object
    Set result = CreateObject("Scripting.Dictionary")
    Dim rs As Object
    Set rs = DbQuery("SELECT * FROM " & collection)
    Do While Not rs.EOF
        Dim obj As Object
        Set obj = g_CollMgr.CreateEmpty(collection)
        RemplirDepuisRecordset obj, rs, collection
        result.Add obj.Id, obj
        rs.MoveNext
    Loop
    rs.Close
    Set LoadSql = result
End Function

Private Sub RemplirDepuisRecordset(ByVal obj As Object, ByVal rs As Object, ByVal collection As String)
    Dim props As Variant, i As Long
    props = GetDomainProperties(collection)
    For i = LBound(props) To UBound(props)
        On Error Resume Next
        Dim v As Variant
        ' Cas spécial : Collection (VBA) ↔ CollectionName (SQL)
        If CStr(props(i)) = "Collection" Then
            v = rs("CollectionName")
        Else
            v = rs(CStr(props(i)))
        End If
        If Not IsNull(v) Then
            ' Conversion date si nécessaire
            If IsDateLike(collection, CStr(props(i))) And Len(CStr(v)) > 0 Then
                v = CDate(v)
            End If
            CallByName obj, CStr(props(i)), VbLet, v
        End If
        Err.Clear
    Next i
End Sub

Public Sub Save(ByVal obj As Object, ByVal collection As String)
    If g_Config.DataMode = "SQL" Then
        SaveSql obj, collection
    Else
        SaveXml obj, collection
    End If
End Sub

Private Sub SaveSql(ByVal obj As Object, ByVal collection As String)
    Dim props As Variant, i As Long
    props = GetDomainProperties(collection)

    ' Détection existence
    Dim rs As Object, exists As Boolean
    Set rs = DbQuery("SELECT 1 AS x FROM " & collection & " WHERE Id = ?", Array(obj.Id))
    exists = Not rs.EOF
    rs.Close

    Dim params() As Variant
    ReDim params(LBound(props) To UBound(props))

    If exists Then
        Dim setClause As String
        For i = LBound(props) To UBound(props)
            Dim v As Variant: v = GetPropertyValue(obj, CStr(props(i)))
            If IsDate(v) Then
                If v > 0 Then v = Format$(v, "yyyy-mm-dd hh:nn:ss") Else v = Null
            ElseIf IsEmpty(v) Then
                v = Null
            End If
            setClause = setClause & IIf(Len(setClause) > 0, ", ", "") & props(i) & " = ?"
            params(i) = v
        Next i
        DbExecute "UPDATE " & collection & " SET " & setClause & " WHERE Id = ?", params
        Journaliser "Modification", collection, obj.Id, GetLibelle(obj, collection)
    Else
        Dim cols As String, vals As String
        For i = LBound(props) To UBound(props)
            Dim v2 As Variant: v2 = GetPropertyValue(obj, CStr(props(i)))
            If IsDate(v2) Then
                If v2 > 0 Then v2 = Format$(v2, "yyyy-mm-dd hh:nn:ss") Else v2 = Null
            ElseIf IsEmpty(v2) Then
                v2 = Null
            End If
            cols = cols & IIf(Len(cols) > 0, ", ", "") & props(i)
            vals = vals & IIf(Len(vals) > 0, ", ", "") & "?"
            params(i) = v2
        Next i
        DbExecute "INSERT INTO " & collection & " (" & cols & ") VALUES (" & vals & ")", params
        Journaliser "Création", collection, obj.Id, GetLibelle(obj, collection)
    End If
End Sub

Public Sub DeleteById(ByVal id As String, ByVal collection As String)
    If g_Config.DataMode = "SQL" Then
        ' Récupérer le libellé avant suppression pour le journal
        Dim rs As Object, libelle As String
        Set rs = DbQuery("SELECT * FROM " & collection & " WHERE Id = ?", Array(id))
        If Not rs.EOF Then
            Dim tmp As Object
            Set tmp = g_CollMgr.CreateEmpty(collection)
            RemplirDepuisRecordset tmp, rs, collection
            libelle = GetLibelle(tmp, collection)
        End If
        rs.Close
        DbExecute "DELETE FROM " & collection & " WHERE Id = ?", Array(id)
        Journaliser "Suppression", collection, id, libelle
    Else
        DeleteXmlById id, collection
    End If
End Sub
