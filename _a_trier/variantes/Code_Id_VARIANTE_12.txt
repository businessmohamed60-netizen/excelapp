Private Sub Class_Initialize()
    Set m_Mesures = CreateObject("Scripting.Dictionary")
    Set m_OperateurNoms = CreateObject("Scripting.Dictionary")
    Set m_PieceReferences = CreateObject("Scripting.Dictionary")
    m_NbPieces = 10: m_NbOperateurs = 3: m_NbEssais = 3
End Sub

Public Sub InitNew()
    m_Id = NewGuid()
    m_DateCreation = Now
End Sub

Public Sub AssignReference()
    Dim dom As Object, root As Object, yr As String
    Dim fname As String, item As Object, numeroNode As Object
    Dim maxN As Long, parts() As String, n As Long
    fname = DataPath() & "msa.xml"
    yr = Format$(Year(Date), "0000")
    If Dir(fname) <> "" Then
        Set dom = CreateObject("MSXML2.DOMDocument.6.0")
        dom.async = False
        If dom.Load(fname) Then
            Set root = dom.documentElement
            For Each item In root.ChildNodes
                Set numeroNode = item.selectSingleNode("Reference")
                If Not numeroNode Is Nothing Then
                    parts = Split(numeroNode.Text, "-")
                    If UBound(parts) >= 2 Then
                        If parts(1) = yr Then
                            n = CLng(Val(parts(2)))
                            If n > maxN Then maxN = n
                        End If
                    End If
                End If
            Next item
        End If
    End If
    m_Reference = "MSA-" & yr & "-" & Format$(maxN + 1, "000")
End Sub

Public Function CleMesure(ByVal op As Long, ByVal piece As Long, ByVal essai As Long) As String
    CleMesure = op & "_" & piece & "_" & essai
End Function

Public Sub SetMesure(ByVal op As Long, ByVal piece As Long, ByVal essai As Long, ByVal val As Double)
    m_Mesures(CleMesure(op, piece, essai)) = val
End Sub

Public Function GetMesure(ByVal op As Long, ByVal piece As Long, ByVal essai As Long) As Double
    Dim k As String
    k = CleMesure(op, piece, essai)
    If m_Mesures.Exists(k) Then GetMesure = m_Mesures(k)
End Function

Public Function MoyenneOpPiece(ByVal op As Long, ByVal piece As Long) As Double
    Dim i As Long, s As Double
    For i = 1 To m_NbEssais: s = s + GetMesure(op, piece, i): Next i
    MoyenneOpPiece = s / m_NbEssais
End Function

Public Function MoyennePiece(ByVal piece As Long) As Double
    Dim op As Long, s As Double
    For op = 1 To m_NbOperateurs: s = s + MoyenneOpPiece(op, piece): Next op
    MoyennePiece = s / m_NbOperateurs
End Function

Public Function MoyenneOperateur(ByVal op As Long) As Double
    Dim piece As Long, s As Double
    For piece = 1 To m_NbPieces: s = s + MoyenneOpPiece(op, piece): Next piece
    MoyenneOperateur = s / m_NbPieces
End Function

Public Function MoyenneGenerale() As Double
    Dim op As Long, s As Double
    For op = 1 To m_NbOperateurs: s = s + MoyenneOperateur(op): Next op
    MoyenneGenerale = s / m_NbOperateurs
End Function

Public Function EtendueMoyenneEssais() As Double
    Dim op As Long, piece As Long, i As Long
    Dim mn As Double, mx As Double, s As Double, n As Long
    For op = 1 To m_NbOperateurs
        For piece = 1 To m_NbPieces
            mn = GetMesure(op, piece, 1): mx = mn
            For i = 2 To m_NbEssais
                If GetMesure(op, piece, i) < mn Then mn = GetMesure(op, piece, i)
                If GetMesure(op, piece, i) > mx Then mx = GetMesure(op, piece, i)
            Next i
            s = s + (mx - mn): n = n + 1
        Next piece
    Next op
    If n > 0 Then EtendueMoyenneEssais = s / n
End Function

Public Function EV() As Double
    EV = EtendueMoyenneEssais() / 1.693
End Function

Public Function AV() As Double
    Dim maxOp As Double, minOp As Double, i As Long, xbar As Double, inside As Double
    maxOp = MoyenneOperateur(1): minOp = maxOp
    For i = 2 To m_NbOperateurs
        If MoyenneOperateur(i) > maxOp Then maxOp = MoyenneOperateur(i)
        If MoyenneOperateur(i) < minOp Then minOp = MoyenneOperateur(i)
    Next i
    xbar = maxOp - minOp
    inside = (xbar / 1.693) ^ 2 - (EV() ^ 2 / (m_NbPieces * m_NbEssais))
    If inside > 0 Then AV = Sqr(inside)
End Function

Public Function GRR() As Double
    GRR = Sqr(EV() ^ 2 + AV() ^ 2)
End Function

Public Function PV() As Double
    Dim arr() As Double, j As Long
    ReDim arr(0 To m_NbPieces - 1)
    For j = 1 To m_NbPieces: arr(j - 1) = MoyennePiece(j): Next j
    PV = EcartType(arr, True)
    If PV = 0 Then PV = 0.0001
End Function

Public Function TotalVariation() As Double
    TotalVariation = Sqr(GRR() ^ 2 + PV() ^ 2)
End Function

Public Function PourcentEV() As Double
    If TotalVariation() > 0 Then PourcentEV = 100 * EV() / TotalVariation()
End Function
Public Function PourcentAV() As Double
    If TotalVariation() > 0 Then PourcentAV = 100 * AV() / TotalVariation()
End Function
Public Function PourcentGRR() As Double
    If TotalVariation() > 0 Then PourcentGRR = 100 * GRR() / TotalVariation()
End Function
Public Function PourcentPV() As Double
    If TotalVariation() > 0 Then PourcentPV = 100 * PV() / TotalVariation()
End Function

Public Function NombreDistinctCategorie() As Double
    If GRR() > 0 Then NombreDistinctCategorie = 1.41 * (PV() / GRR())
End Function

Public Function Verdict() As String
    Dim g As Double: g = PourcentGRR()
    If g < 10 Then Verdict = "Acceptable" ElseIf g <= 30 Then Verdict = "Conditionnel" Else Verdict = "Rejeté"
End Function

Public Function Validate() As Boolean
    If Len(m_ClientId) = 0 Then Exit Function
    If m_Tolerance <= 0 Then Exit Function
    If m_Mesures.Count < m_NbOperateurs * m_NbPieces * m_NbEssais Then Exit Function
    Validate = True
End Function
