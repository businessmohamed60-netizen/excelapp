Public Sub InitNew()
    m_Id = NewGuid()
    m_DateCreation = Now
    m_DateDetection = Date
    m_Statut = "Ouverte"
    m_Classification = "Mineur"
    m_Gravite = 2
    m_Source = "Interne"
End Sub

Public Sub AssignNumero()
    Dim dom As Object, root As Object, yr As String
    Dim fname As String, item As Object, itemYr As String, maxN As Long
    fname = DataPath() & "nc.xml"
    yr = Format$(Year(m_DateDetection), "0000")
    If Dir(fname) <> "" Then
        Set dom = CreateObject("MSXML2.DOMDocument.6.0")
        dom.async = False
        If dom.Load(fname) Then
            Set root = dom.documentElement
            For Each item In root.ChildNodes
                Dim numeroNode As Object
                Set numeroNode = item.selectSingleNode("Numero")
                If Not numeroNode Is Nothing Then
                    Dim parts() As String
                    parts = Split(numeroNode.Text, "-")
                    If UBound(parts) >= 2 Then
                        itemYr = parts(1)
                        If itemYr = yr Then
                            Dim n As Long: n = CLng(Val(parts(2)))
                            If n > maxN Then maxN = n
                        End If
                    End If
                End If
            Next item
        End If
    End If
    m_Numero = "NC-" & yr & "-" & Format$(maxN + 1, "000")
End Sub

Public Function Validate() As Boolean
    Dim errs As New Collection
    If Len(Trim$(m_ClientId)) = 0 Then errs.Add "Client obligatoire"
    If Len(Trim$(m_Description)) = 0 Then errs.Add "Description obligatoire"
    If Not IsDate(m_DateDetection) Then errs.Add "Date de détection invalide"
    If m_Statut = "Clôturée" And Not IsDate(m_DateCloture) Then errs.Add "Date de clôture requise"
    Validate = (errs.Count = 0)
End Function

Public Function DisplayName() As String
    DisplayName = m_Numero & " — " & m_Classification & " — " & Left$(m_Description, 50)
End Function
