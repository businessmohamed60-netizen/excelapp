    ' AMDEC
    Dim amdecs As Object, lignesAmdec As Object
    Dim nbAmdecs As Long, nbLignesCritiques As Long, nbActionsAmdecRetard As Long
    Set amdecs = g_CollMgr.LoadCollection("amdec")
    Set lignesAmdec = g_CollMgr.LoadCollection("ligneamdec")
    For Each k In amdecs.Keys
        If amdecs(k).Statut = "Approuvée" Then nbAmdecs = nbAmdecs + 1
    Next k
    For Each k In lignesAmdec.Keys
        If lignesAmdec(k).Criticite = "Critique" And lignesAmdec(k).StatutAction <> "Clôturée" Then
            nbLignesCritiques = nbLignesCritiques + 1
        End If
        If lignesAmdec(k).EstEnRetard() Then nbActionsAmdecRetard = nbActionsAmdecRetard + 1
    Next k

    result.Add EvaluerKpi("AMDEC approuvées", nbAmdecs, 1, "Haut", "")
    result.Add EvaluerKpi("Lignes AMDEC critiques ouvertes", nbLignesCritiques, 0, "Bas", "")
    result.Add EvaluerKpi("Actions AMDEC en retard", nbActionsAmdecRetard, 0, "Bas", "")
