Option Explicit

Private m_Id As String
Private m_Reference As String
Private m_ClientId As String
Private m_TypeAmdec As String
Private m_Piece As String
Private m_Processus As String
Private m_Equipe As String
Private m_DateCreation As Date
Private m_Lignes As Collection

Public Property Let Id(ByVal v As String): m_Id = v: End Property

Public Property Let Reference(ByVal v As String): m_Reference = v: End Property

Public Property Get Equipe() As String: Equipe = m_Equipe: End Property
Public Property Let Equipe(ByVal v As String): m_Equipe = v: End Property
Public Property Get DateCreation() As Date: DateCreation = m_DateCreation: End Property
Public Property Let DateCreation(ByVal v As Date): m_DateCreation = v: End Property
Public Property Get Lignes() As Collection: Set Lignes = m_Lignes: End Property

Private Sub Class_Initialize()
    Set m_Lignes = New Collection
End Sub

Public Sub InitNew()
    m_Id = NewGuid()
    m_DateCreation = Now
    m_TypeAmdec = "Processus"
End Sub

Public Sub AssignReference()
    Dim dom As Object, root As Object, yr As String
    Dim fname As String, item As Object, numeroNode As Object
    Dim maxN As Long, parts() As String, n As Long
    fname = DataPath() & "amdec.xml"
    yr = Format$(Year(Date), "0000")
    If Dir(fname) <> "" Then
        Set dom = CreateObject("MSXML2.DOMDocument.6.0")
        dom.async = False
        If dom.Load(fname) Then
            Set root = dom.documentElement
            For Each item In root.ChildNodes
                Set numeroNode = item.selectSingleNode("Reference")
                If Not numeroNode Is Nothing Then
                    parts = Split(numeroNode.Text, "-")
                    If UBound(parts) >= 2 Then
                        If parts(1) = yr Then
                            n = CLng(Val(parts(2)))
                            If n > maxN Then maxN = n
                        End If
                    End If
                End If
            Next item
        End If
    End If
    m_Reference = "AMDEC-" & yr & "-" & Format$(maxN + 1, "000")
End Sub

Public Function RPNMax() As Long
    Dim l As CLigneAmdec, mx As Long
    For Each l In m_Lignes
        If l.RPN() > mx Then mx = l.RPN()
    Next l
    RPNMax = mx
End Function

Public Function NbActionsH() As Long
    Dim l As CLigneAmdec, n As Long
    For Each l In m_Lignes
        If l.ActionPriority() = "H" Then n = n + 1
    Next l
    NbActionsH = n
End Function

Public Function Validate() As Boolean
    If Len(m_ClientId) = 0 Then Exit Function
    If Len(Trim$(m_Piece)) = 0 Then Exit Function
    If m_Lignes.Count = 0 Then Exit Function
    Validate = True
End Function
