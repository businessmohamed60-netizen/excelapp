Attribute VB_Name = "M_Migration"
Option Explicit

' ============================================================
' M_Migration — Migration one-shot XML → SQLite
' ============================================================

Public Sub MigrerTout()
    On Error GoTo ErrHandler

    ' 1. Sauvegarde de sécurité du .db s'il existe
    If Dir(g_Config.DbPath) <> "" Then
        Dim backup As String
        backup = g_Config.DbPath & ".backup_" & Format$(Now, "yyyymmdd_hhnnss")
        FileCopy g_Config.DbPath, backup
        Log LogInfo, "M_Migration", "Sauvegarde pré-migration : " & backup
    End If

    ' 2. Ouvrir + créer schéma
    If Not DbOpen() Then
        ShowError "Migration", "Impossible d'ouvrir la base SQLite."
        Exit Sub
    End If
    CreerSchema

    ' 3. Migrer chaque collection dans l'ordre des dépendances
    Dim tables As Variant, i As Long, total As Long, nb As Long
    tables = Array( _
        "clients", "audits", "constats", "nc", "capa", "huitd", _
        "auditprogramme", "document", "exigence", _
        "amdec", "ligneamdec", "plancontrole", "ligneplan", "caracteristique", _
        "moyenmesure", "etalonnage", "etudemsa", "lignemsa", _
        "etudecapabilite", "cartecontrole", "lignecarte", "audittrail")

    For i = LBound(tables) To UBound(tables)
        nb = MigrerCollection(CStr(tables(i)))
        If nb >= 0 Then
            total = total + nb
            Log LogInfo, "M_Migration", tables(i) & " : " & nb & " enregistrements"
        Else
            Log LogError, "M_Migration", "Échec migration " & tables(i)
        End If
    Next i

    ' 4. Basculer le mode
    g_Config.DataMode = "SQL"
    SauvegarderConfig g_Config

    ShowInfo "Migration", total & " enregistrements migrés vers SQLite." & vbCrLf & _
        "Mode SQL activé. Redémarrez Excel pour prendre en compte."
    Exit Sub
ErrHandler:
    DbRollback
    LogEx "M_Migration", "MigrerTout", Err
    ShowError "Migration", Err.Description
End Sub

Public Function MigrerCollection(ByVal collection As String) As Long
    On Error GoTo ErrHandler
    If Not DbOpen() Then Exit Function

    ' Vider la table cible (migration = remplacement)
    DbExecute "DELETE FROM " & collection

    Dim obj As Object, k As Variant, n As Long
    Set obj = g_CollMgr.LoadCollection(collection)
    If obj Is Nothing Then MigrerCollection = 0: Exit Function

    DbBegin
    For Each k In obj.Keys
        InsererObjet obj(k), collection
        n = n + 1
    Next k
    DbCommit

    MigrerCollection = n
    Exit Function
ErrHandler:
    DbRollback
    LogEx "M_Migration", "MigrerCollection", Err
    MigrerCollection = -1
End Function

Private Sub InsererObjet(ByVal obj As Object, ByVal collection As String)
    Dim props As Variant, i As Long, cols As String, vals As String
    Dim params() As Variant
    props = GetDomainProperties(collection)
    If IsEmpty(props) Then Exit Sub

    ReDim params(LBound(props) To UBound(props))
    For i = LBound(props) To UBound(props)
        Dim v As Variant
        v = GetPropertyValue(obj, CStr(props(i)))
        If IsDate(v) Then
            If v > 0 Then v = Format$(v, "yyyy-mm-dd hh:nn:ss") Else v = Null
        ElseIf IsEmpty(v) Then
            v = Null
        End If
        cols = cols & IIf(Len(cols) > 0, ", ", "") & CStr(props(i))
        vals = vals & IIf(Len(vals) > 0, ", ", "") & "?"
        params(i) = v
    Next i
    DbExecute "INSERT INTO " & collection & " (" & cols & ") VALUES (" & vals & ")", params
End Sub

Private Function GetPropertyValue(ByVal obj As Object, ByVal propName As String) As Variant
    On Error Resume Next
    GetPropertyValue = CallByName(obj, propName, VbGet)
    If Err.Number <> 0 Then GetPropertyValue = Empty
    Err.Clear
End Function
