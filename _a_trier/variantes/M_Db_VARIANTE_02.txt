Attribute VB_Name = "M_Db"
Option Explicit

' ============================================================
' M_Db — Couche d'accès SQLite via ADODB + SQLite3 ODBC Driver
' Phase 8 — décision : SQLite uniquement
' ============================================================

Private m_Conn As Object            ' ADODB.Connection
Private m_Transaction As Boolean
Private m_BusyTimeout As Long

Public Function DbOpen() As Boolean
    On Error GoTo ErrHandler
    If Not m_Conn Is Nothing Then
        If m_Conn.State = 1 Then DbOpen = True: Exit Function
    End If

    ' Vérifier que le driver est présent
    If Not DriverSqlitePresent() Then
        Err.Raise vbObjectError + 5000, "M_Db", _
            "Driver SQLite3 ODBC non installé. Téléchargez-le sur https://www.ch-werner.de/sqliteodbc/"
    End If

    ' Créer le dossier si nécessaire
    Dim dossier As String
    dossier = Left$(g_Config.DbPath, InStrRev(g_Config.DbPath, "\"))
    If Dir(dossier, vbDirectory) = "" Then MkDir dossier

    Set m_Conn = CreateObject("ADODB.Connection")
    m_Conn.ConnectionString = "DRIVER={SQLite3 ODBC Driver};Database=" & g_Config.DbPath & ";"
    m_Conn.CommandTimeout = 30
    m_Conn.Open

    ' Options SQLite recommandées
    m_Conn.Execute "PRAGMA foreign_keys = ON;"
    m_Conn.Execute "PRAGMA journal_mode = WAL;"      ' Write-Ahead Logging : meilleure concurrence
    m_Conn.Execute "PRAGMA synchronous = NORMAL;"    ' compromis performance/sécurité
    m_Conn.Execute "PRAGMA busy_timeout = 5000;"     ' attendre 5s en cas de verrou

    m_BusyTimeout = 5000
    DbOpen = True
    Log LogInfo, "M_Db", "Connexion SQLite ouverte : " & g_Config.DbPath
    Exit Function
ErrHandler:
    LogEx "M_Db", "DbOpen", Err
    DbOpen = False
End Function

Public Sub DbClose()
    On Error Resume Next
    If m_Transaction And Not m_Conn Is Nothing Then m_Conn.RollbackTrans
    If Not m_Conn Is Nothing Then
        If m_Conn.State = 1 Then m_Conn.Close
    End If
    Set m_Conn = Nothing
    m_Transaction = False
End Sub

' ----- Vérification du driver -----

Public Function DriverSqlitePresent() As Boolean
    On Error Resume Next
    Dim rs As Object
    Set rs = CreateObject("ADODB.Connection")
    ' Test simple : essayer d'ouvrir une connexion mémoire
    rs.Open "DRIVER={SQLite3 ODBC Driver};Database=:memory:;"
    DriverSqlitePresent = (Err.Number = 0)
    If Not rs Is Nothing Then
        If rs.State = 1 Then rs.Close
    End If
    Err.Clear
End Function

' ----- Exécution -----

Public Function DbExecute(ByVal sql As String, Optional ByVal params As Variant) As Long
    On Error GoTo ErrHandler
    If Not DbOpen() Then Exit Function

    Dim cmd As Object
    Set cmd = CreateObject("ADODB.Command")
    Set cmd.ActiveConnection = m_Conn
    cmd.CommandText = sql
    cmd.CommandType = 1 ' adCmdText

    If Not IsMissing(params) Then
        Dim i As Long
        For i = LBound(params) To UBound(params)
            cmd.Parameters.Append CreerParametre(cmd, params(i))
        Next i
    End If

    cmd.Execute , , 128 ' adExecuteNoRecords
    DbExecute = 1
    Exit Function
ErrHandler:
    LogEx "M_Db", "DbExecute", Err
    Err.Raise Err.Number, "M_Db.DbExecute", Err.Description & vbCrLf & "SQL: " & sql
End Function

Public Function DbQuery(ByVal sql As String, Optional ByVal params As Variant) As Object
    On Error GoTo ErrHandler
    If Not DbOpen() Then Exit Function

    Dim cmd As Object, rs As Object
    Set cmd = CreateObject("ADODB.Command")
    Set cmd.ActiveConnection = m_Conn
    cmd.CommandText = sql
    cmd.CommandType = 1

    If Not IsMissing(params) Then
        Dim i As Long
        For i = LBound(params) To UBound(params)
            cmd.Parameters.Append CreerParametre(cmd, params(i))
        Next i
    End If

    Set rs = cmd.Execute
    rs.ActiveConnection = Nothing     ' détacher pour permettre DbClose
    Set DbQuery = rs
    Exit Function
ErrHandler:
    LogEx "M_Db", "DbQuery", Err
    Err.Raise Err.Number, "M_Db.DbQuery", Err.Description & vbCrLf & "SQL: " & sql
End Function

' Crée un paramètre typé selon la valeur
Private Function CreerParametre(ByVal cmd As Object, ByVal v As Variant) As Object
    Dim p As Object
    If IsNull(v) Or IsEmpty(v) Then
        Set p = cmd.CreateParameter("p", 200, 1, 255, Null)    ' adVarChar, adParamInput
    ElseIf VarType(v) = vbBoolean Then
        Set p = cmd.CreateParameter("p", 2, 1, , IIf(v, 1, 0))  ' adInteger
    ElseIf IsDate(v) Then
        Set p = cmd.CreateParameter("p", 200, 1, 19, Format$(v, "yyyy-mm-dd hh:nn:ss"))
    ElseIf IsNumeric(v) Then
        If InStr(1, CStr(v), ".") > 0 Then
            Set p = cmd.CreateParameter("p", 5, 1, , CDbl(v))    ' adDouble
        Else
            Set p = cmd.CreateParameter("p", 3, 1, , CLng(v))    ' adInteger
        End If
    Else
        Set p = cmd.CreateParameter("p", 200, 1, 4000, CStr(v)) ' adVarChar
    End If
    Set CreerParametre = p
End Function

' ----- Transactions -----

Public Sub DbBegin()
    If Not DbOpen() Then Exit Sub
    ' BEGIN IMMEDIATE : verrou d'écriture immédiat (évite les deadlocks SQLite)
    m_Conn.Execute "BEGIN IMMEDIATE;"
    m_Transaction = True
End Sub

Public Sub DbCommit()
    If m_Transaction And Not m_Conn Is Nothing Then
        m_Conn.Execute "COMMIT;"
        m_Transaction = False
    End If
End Sub

Public Sub DbRollback()
    If m_Transaction And Not m_Conn Is Nothing Then
        On Error Resume Next
        m_Conn.Execute "ROLLBACK;"
        m_Transaction = False
        On Error GoTo 0
    End If
End Sub

' ----- Utilitaires -----

Public Function DbTest() As Boolean
    On Error GoTo ErrHandler
    If Not DbOpen() Then DbTest = False: Exit Function
    Dim rs As Object
    Set rs = DbQuery("SELECT 1 AS test")
    If Not rs.EOF Then DbTest = True
    rs.Close
    Exit Function
ErrHandler:
    DbTest = False
End Function

Public Function DbTables() As Collection
    Dim result As New Collection, rs As Object
    On Error Resume Next
    Set rs = DbQuery("SELECT name FROM sqlite_master WHERE type='table' AND name NOT LIKE 'sqlite_%' ORDER BY name")
    Do While Not rs.EOF
        result.Add rs("name")
        rs.MoveNext
    Loop
    rs.Close
    Set DbTables = result
End Function

' Compte les enregistrements d'une table
Public Function DbCount(ByVal table As String) As Long
    On Error Resume Next
    Dim rs As Object
    Set rs = DbQuery("SELECT COUNT(*) AS n FROM " & table)
    If Not rs.EOF Then DbCount = CLng(rs("n"))
    rs.Close
End Function

' Sauvegarde atomique du fichier .db (copie avec vérification)
Public Function DbBackup(ByVal destPath As String) As Boolean
    On Error GoTo ErrHandler
    If Not DbOpen() Then Exit Function
    ' SQLite : VACUUM INTO copie de manière atomique
    If Dir(destPath) <> "" Then Kill destPath
    DbExecute "VACUUM INTO '" & Replace(destPath, "'", "''") & "';"
    DbBackup = True
    Log LogInfo, "M_Db", "Sauvegarde SQLite : " & destPath
    Exit Function
ErrHandler:
    LogEx "M_Db", "DbBackup", Err
    DbBackup = False
End Function

' Restaure une sauvegarde (ferme la connexion, copie, rouvre)
Public Function DbRestore(ByVal srcPath As String) As Boolean
    On Error GoTo ErrHandler
    If Dir(srcPath) = "" Then Exit Function
    DbClose
    FileCopy srcPath, g_Config.DbPath
    DbRestore = DbOpen()
    Log LogInfo, "M_Db", "Restauration SQLite depuis : " & srcPath
    Exit Function
ErrHandler:
    LogEx "M_Db", "DbRestore", Err
    DbRestore = False
End Function
