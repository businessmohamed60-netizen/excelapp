' ============================================================
' RAFRAÎCHISSEMENT GLOBAL
' ============================================================
Public Sub RefreshAll()
    On Error GoTo ErrHandler
    m_DernierRefresh = Now
    lblDateHeure.Caption = "Actualisé : " & Format$(Now, "dd/mm/yyyy hh:nn")
    RefreshKpis
    RefreshAlertes
    RefreshActivites
    RefreshGraphiques
    RefreshHeatmap
    Exit Sub
ErrHandler:
    LogEx "FMain", "RefreshAll", Err
End Sub

' ============================================================
' KPI — 12 cartes
' ============================================================
Private Sub RefreshKpis()
    Set m_Kpis = CalculerTousKpis()
    Dim i As Long
    For i = 1 To m_Kpis.Count
        AppliquerKpiSurCarte i, m_Kpis(i)
    Next i
End Sub

Private Sub AppliquerKpiSurCarte(ByVal index As Long, ByVal kpi As Object)
    On Error Resume Next
    Dim lblChiffre As Object, lblLabel As Object, lblIcone As Object, fraCarte As Object
    Set lblChiffre = Me.Controls("lblKpi" & index)
    Set lblLabel = Me.Controls("lblKpi" & index & "Lbl")
    Set lblIcone = Me.Controls("lblKpi" & index & "Ico")
    Set fraCarte = Me.Controls("fraKpi" & index)
    lblChiffre.Caption = FormaterValeurKpi(kpi("Valeur"))
    lblChiffre.Font.Name = PoliceChiffres()
    lblChiffre.Font.Size = TailleKpiChiffre()
    lblChiffre.Font.Bold = True
    lblChiffre.ForeColor = CouleurPourNiveau(kpi("Niveau"))
    lblLabel.Caption = kpi("Label")
    lblLabel.Font.Name = PoliceCorps()
    lblLabel.Font.Size = TailleKpiLabel()
    lblLabel.ForeColor = TexteSecondaire()
    lblIcone.Caption = kpi("Icone")
    lblIcone.Font.Size = 20
    If Not fraCarte Is Nothing Then
        fraCarte.BackColor = CouleurCarteFond()
        fraCarte.BorderStyle = 1
        fraCarte.BorderColor = CouleurPourNiveau(kpi("Niveau"))
        fraCarte.SpecialEffect = 0
    End If
    Err.Clear
    On Error GoTo 0
End Sub

Private Function FormaterValeurKpi(ByVal v As Variant) As String
    If IsNumeric(v) Then
        Dim d As Double: d = CDbl(v)
        If d = Int(d) Then FormaterValeurKpi = Format$(d, "#,##0") Else FormaterValeurKpi = Format$(d, "0.0")
    Else
        FormaterValeurKpi = CStr(v)
    End If
End Function

' ============================================================
' ALERTES
' ============================================================
Private Sub RefreshAlertes()
    lstAlertes.Clear
    Dim notifs As Collection: Set notifs = CollecterNotifications()
    Dim arr() As Object, i As Long, j As Long
    If notifs.Count > 0 Then
        ReDim arr(0 To notifs.Count - 1)
        For i = 1 To notifs.Count: Set arr(i - 1) = notifs(i): Next i
        Dim tmp As Object
        For i = 0 To UBound(arr) - 1
            For j = i + 1 To UBound(arr)
                If arr(j)("Severite") > arr(i)("Severite") Then Set tmp = arr(i): Set arr(i) = arr(j): Set arr(j) = tmp
            Next j
        Next i
    End If
    Dim limit As Long: limit = IIf(notifs.Count > 8, 8, notifs.Count)
    For i = 0 To limit - 1
        Dim n As Object: Set n = arr(i)
        Dim sevStr As String
        Select Case n("Severite")
            Case NotifCritique: sevStr = "🔴"
            Case NotifAvertissement: sevStr = "🟠"
            Case Else: sevStr = "🔵"
        End Select
        lstAlertes.AddItem sevStr
        lstAlertes.List(lstAlertes.ListCount - 1, 1) = n("Categorie")
        lstAlertes.List(lstAlertes.ListCount - 1, 2) = n("Titre")
    Next i
End Sub

' ============================================================
' ACTIVITÉ RÉCENTE
' ============================================================
Private Sub RefreshActivites()
    lstActivites.Clear
    Dim audits As Object, ncs As Object, capas As Object
    Set audits = g_CollMgr.LoadCollection("audits")
    Set ncs = g_CollMgr.LoadCollection("nc")
    Set capas = g_CollMgr.LoadCollection("capa")
    Dim items As Object: Set items = CreateObject("Scripting.Dictionary")
    Dim k As Variant, entry As String
    For Each k In audits.Keys
        entry = Format$(audits(k).DateCreation, "yyyy-mm-dd hh:nn") & " | 🔍 Audit | " & audits(k).Processus & " (" & audits(k).Statut & ")"
        items.Add "A" & k, entry
    Next k
    For Each k In ncs.Keys
        entry = Format$(ncs(k).DateCreation, "yyyy-mm-dd hh:nn") & " | ⚠️ NC | " & ncs(k).Numero & " — " & Left$(ncs(k).Description, 40)
        items.Add "N" & k, entry
    Next k
    For Each k In capas.Keys
        entry = Format$(capas(k).DateCreation, "yyyy-mm-dd hh:nn") & " | ✅ CAPA | " & capas(k).Numero & " (" & capas(k).Statut & ")"
        items.Add "C" & k, entry
    Next k
    Dim keys As Variant, i As Long, j As Long, tmp As String
    keys = items.Keys
    For i = LBound(keys) To UBound(keys) - 1
        For j = i + 1 To UBound(keys)
            If items(keys(j)) > items(keys(i)) Then tmp = keys(i): keys(i) = keys(j): keys(j) = tmp
        Next j
    Next i
    Dim count As Long
    For i = LBound(keys) To UBound(keys)
        lstActivites.AddItem items(keys(i))
        count = count + 1
        If count >= 12 Then Exit For
    Next i
    If lstActivites.ListCount = 0 Then lstActivites.AddItem "(aucune activité récente)"
End Sub

' ============================================================
' GRAPHIQUES — 4 images PNG
' ============================================================
Private Sub RefreshGraphiques()
    On Error Resume Next
    Set m_GraphPaths = GenererGraphiques()
    If m_GraphPaths Is Nothing Then Exit Sub
    ChargerImage "imgGraph1", m_GraphPaths("Graph1")
    ChargerImage "imgGraph2", m_GraphPaths("Graph2")
    ChargerImage "imgGraph3", m_GraphPaths("Graph3")
    ChargerImage "imgGraph4", m_GraphPaths("Graph4")
    Err.Clear
    On Error GoTo 0
End Sub

Private Sub ChargerImage(ByVal nomControle As String, ByVal chemin As String)
    On Error Resume Next
    If Dir(chemin) = "" Then Exit Sub
    Me.Controls(nomControle).Picture = LoadPicture(chemin)
    Me.Controls(nomControle).PictureSizeMode = fmPictureSizeModeZoom
    Err.Clear
    On Error GoTo 0
End Sub

' ============================================================
' HEATMAP — 12 semaines x 7 jours
' ============================================================
Private Sub InitializeHeatmap()
    ' Créer dynamiquement 84 labels dans fraHeatmap si absents
    On Error Resume Next
    Dim i As Long, j As Long, ctrlName As String
    For i = 1 To 84
        ctrlName = "lblHeat" & i
        If Not ControlExists(ctrlName) Then
            Dim lbl As MSForms.Label
            Set lbl = fraHeatmap.Controls.Add("Forms.Label.1", ctrlName, True)
            lbl.Width = 18: lbl.Height = 18
            lbl.Caption = ""
            lbl.BorderStyle = 1
            lbl.BorderColor = CouleurBordure()
            ' Position en grille 12 x 7
            Dim semaine As Long, jour As Long
            semaine = ((i - 1) \ 7) + 1
            jour = ((i - 1) Mod 7) + 1
            lbl.Left = 10 + (semaine - 1) * 20
            lbl.Top = 10 + (jour - 1) * 20
        End If
    Next i
    Err.Clear
    On Error GoTo 0
End Sub

Private Function ControlExists(ByVal nom As String) As Boolean
    On Error Resume Next
    Dim c As Object
    Set c = Me.Controls(nom)
    ControlExists = Not (c Is Nothing)
    Err.Clear
    On Error GoTo 0
End Function

Private Sub RefreshHeatmap()
    On Error Resume Next
    Dim activite As Object: Set activite = CalculerActiviteParJour(12)
    Dim maxAct As Long: maxAct = ActiviteMax(activite)
    Dim dateRef As Date: dateRef = Date - 83
    Dim i As Long, semaine As Long, jour As Long, dte As Date, cle As String
    Dim lbl As Object
    ' Pour chaque cellule, calculer le jour correspondant
    For i = 1 To 84
        semaine = ((i - 1) \ 7)
        jour = ((i - 1) Mod 7)
        dte = dateRef + semaine * 7 + jour
        cle = FormatIsoDate(dte)
        Dim valeur As Long: valeur = 0
        If activite.Exists(cle) Then valeur = activite(cle)
        Set lbl = fraHeatmap.Controls("lblHeat" & i)
        lbl.BackColor = CouleurHeatmap(valeur, maxAct)
        lbl.ControlTipText = Format$(dte, "dd/mm/yyyy") & " — " & valeur & " événement(s)"
    Next i
    Err.Clear
    On Error GoTo 0
End Sub

' ============================================================
' RAFRAÎCHISSEMENT AUTO (toutes les 5 min)
' ============================================================
Private Sub ProgrammerRafraichissementAuto()
    On Error Resume Next
    Application.OnTime Now + TimeValue("00:05:00"), "RafraichirDashboard"
End Sub

Public Sub RafraichirDashboard()
    On Error Resume Next
    If Not Me Is Nothing Then
        If Me.Visible Then RefreshAll
        ProgrammerRafraichissementAuto
    End If
End Sub

    If idx < 0 Then Exit Sub

    If idx + 1 <= notifs.Count Then
        OuvrirObjetDepuisNotif notifs(idx + 1)("Action"), notifs(idx + 1)("ObjetId")
    End If
End Sub
