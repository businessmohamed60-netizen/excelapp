Public Sub InitNew()
    m_Id = NewGuid()
    m_S = 5: m_O = 5: m_D = 5
End Sub

Public Function RPN() As Long: RPN = m_S * m_O * m_D: End Function
Public Function NouveauRPN() As Long
    If m_NouveauS > 0 Then NouveauRPN = m_NouveauS * m_NouveauO * m_NouveauD
End Function
Public Function ActionPriority() As String: ActionPriority = CalculAP(m_S, m_O, m_D): End Function
Public Function NouvelleAP() As String
    If m_NouveauS > 0 Then NouvelleAP = CalculAP(m_NouveauS, m_NouveauO, m_NouveauD)
End Function

Private Function CalculAP(ByVal s As Long, ByVal o As Long, ByVal d As Long) As String
    If s >= 9 Then
        If o >= 2 Or d >= 5 Then CalculAP = "H" Else CalculAP = "M"
    ElseIf s >= 7 Then
        If o >= 4 Then CalculAP = "H" ElseIf o >= 2 Then CalculAP = "M" Else CalculAP = "L"
    ElseIf s >= 4 Then
        If o >= 6 Then CalculAP = "H" ElseIf o >= 4 Then CalculAP = "M" Else CalculAP = "L"
    ElseIf s >= 2 Then
        If o >= 5 Then CalculAP = "M" Else CalculAP = "L"
    Else
        CalculAP = "L"
    End If
End Function
