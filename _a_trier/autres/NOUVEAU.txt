Public Sub RefreshAll()
    On Error GoTo ErrHandler
    Dim clients As Object, audits As Object, constats As Object
    Dim ncs As Object, capas As Object

    Set clients = g_CollMgr.LoadCollection("clients")
    Set audits = g_CollMgr.LoadCollection("audits")
    Set constats = g_CollMgr.LoadCollection("constats")
    Set ncs = g_CollMgr.LoadCollection("nc")
    Set capas = g_CollMgr.LoadCollection("capa")

    lblKpiClients.Caption = CStr(clients.Count)
    lblKpiClientsLbl.Caption = "Clients actifs"

    Dim nbAuditsEnCours As Long, nbAuditsTotal As Long, k As Variant
    For Each k In audits.Keys
        nbAuditsTotal = nbAuditsTotal + 1
        If audits(k).Statut = "En cours" Then nbAuditsEnCours = nbAuditsEnCours + 1
    Next k
    lblKpiAudits.Caption = CStr(nbAuditsEnCours)
    lblKpiAuditsLbl.Caption = "Audits en cours (" & nbAuditsTotal & " au total)"

    Dim nbMajeurs As Long
    For Each k In constats.Keys
        If constats(k).Classification = "Majeur" Then nbMajeurs = nbMajeurs + 1
    Next k
    lblKpiNc.Caption = CStr(nbMajeurs)
    lblKpiNcLbl.Caption = "NC majeures ouvertes"

    Dim nbClotures As Long
    For Each k In audits.Keys
        If audits(k).Statut = "Clôturé" Then nbClotures = nbClotures + 1
    Next k
    If nbAuditsTotal > 0 Then
        lblKpiCapa.Caption = Format$(nbClotures / nbAuditsTotal, "0%")
    Else
        lblKpiCapa.Caption = "—"
    End If
    lblKpiCapaLbl.Caption = "Taux de clôture audits"

    ' NOUVEAU — NC ouvertes
    Dim nbNcOuvertes As Long
    For Each k In ncs.Keys
        If ncs(k).Statut <> "Clôturée" Then nbNcOuvertes = nbNcOuvertes + 1
    Next k
    lblKpiNcOuvertes.Caption = CStr(nbNcOuvertes)
    lblKpiNcOuvertesLbl.Caption = "NC ouvertes"

    ' NOUVEAU — CAPA en retard
    Dim nbCapaRetard As Long
    For Each k In capas.Keys
        If capas(k).EstEnRetard() Then nbCapaRetard = nbCapaRetard + 1
    Next k
    lblKpiCapaRetard.Caption = CStr(nbCapaRetard)
    lblKpiCapaRetardLbl.Caption = "CAPA en retard"

    RefreshActivites audits, constats
    Exit Sub
ErrHandler:
    LogEx "FMain", "RefreshAll", Err
End Sub
