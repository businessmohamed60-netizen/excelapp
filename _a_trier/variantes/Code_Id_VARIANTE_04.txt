Public Sub InitNew()
    m_Id = NewGuid()
    m_DateCreation = Now
    m_DateOuverture = Date
    m_DateEcheance = Date + 30
    m_Statut = "Ouverte"
    m_TypeAction = "Corrective"
    m_Efficacite = "À vérifier"
End Sub

Public Sub AssignNumero()
    Dim dom As Object, root As Object, yr As String
    Dim fname As String, item As Object, numeroNode As Object
    Dim maxN As Long, parts() As String, n As Long
    fname = DataPath() & "capa.xml"
    yr = Format$(Year(m_DateOuverture), "0000")
    If Dir(fname) <> "" Then
        Set dom = CreateObject("MSXML2.DOMDocument.6.0")
        dom.async = False
        If dom.Load(fname) Then
            Set root = dom.documentElement
            For Each item In root.ChildNodes
                Set numeroNode = item.selectSingleNode("Numero")
                If Not numeroNode Is Nothing Then
                    parts = Split(numeroNode.Text, "-")
                    If UBound(parts) >= 2 Then
                        If parts(1) = yr Then
                            n = CLng(Val(parts(2)))
                            If n > maxN Then maxN = n
                        End If
                    End If
                End If
            Next item
        End If
    End If
    m_Numero = "CAPA-" & yr & "-" & Format$(maxN + 1, "000")
End Sub

Public Function EstEnRetard() As Boolean
    EstEnRetard = (m_DateEcheance < Date) And (m_Statut <> "Clôturée")
End Function

Public Function Validate() As Boolean
    If Len(m_NcId) = 0 Then Exit Function
    If Len(Trim$(m_Description)) = 0 Then Exit Function
    If m_DateEcheance < m_DateOuverture Then Exit Function
    Validate = True
End Function
