Public Sub AssignNumero()
    Dim dom As Object, root As Object, yr As String
    Dim fname As String, item As Object, numeroNode As Object
    Dim maxN As Long, parts() As String, n As Long
    fname = DataPath() & "huitd.xml"
    yr = Format$(Year(Date), "0000")
    If Dir(fname) <> "" Then
        Set dom = CreateObject("MSXML2.DOMDocument.6.0")
        dom.async = False
        If dom.Load(fname) Then
            Set root = dom.documentElement
            For Each item In root.ChildNodes
                Set numeroNode = item.selectSingleNode("Numero")
                If Not numeroNode Is Nothing Then
                    parts = Split(numeroNode.Text, "-")
                    If UBound(parts) >= 2 Then
                        If parts(1) = yr Then
                            n = CLng(Val(parts(2)))
                            If n > maxN Then maxN = n
                        End If
                    End If
                End If
            Next item
        End If
    End If
    m_Numero = "8D-" & yr & "-" & Format$(maxN + 1, "000")
End Sub

Public Function EtapeCourante() As String
    If IsDate(m_D8_Date) Then EtapeCourante = "D8" Else
    If IsDate(m_D7_Date) Then EtapeCourante = "D7" Else
    If IsDate(m_D6_Date) Then EtapeCourante = "D6" Else
    If IsDate(m_D5_Date) Then EtapeCourante = "D5" Else
    If IsDate(m_D4_Date) Then EtapeCourante = "D4" Else
    If IsDate(m_D3_Date) Then EtapeCourante = "D3" Else
    If IsDate(m_D2_Date) Then EtapeCourante = "D2" Else
        EtapeCourante = "D1"
End Function

Public Function Validate() As Boolean

    If Len(Trim$(m_D4_CauseRacine)) = 0 Then Exit Function
    Validate = True
End Function
