Public Sub Save(ByVal obj As Object, ByVal collection As String)
    ' ... logique existante ...
    ' Déterminer l'action
    Dim action As String
    If obj.Id = "" Or Not Exists(obj.Id, collection) Then
        action = "Création"
    Else
        action = "Modification"
    End If
    Journaliser action, collection, obj.Id, GetLibelle(obj, collection)
End Sub
