    If idx >= 1 And idx <= 18 Then Element = m_El(idx)

    If idx >= 1 And idx <= 18 Then m_El(idx) = v

Public Sub InitNew()
    m_Id = NewGuid()
    m_DateCreation = Now
    m_NiveauPpap = 3
    m_Statut = "En préparation"
End Sub

Public Sub AssignReference()
    Dim dom As Object, root As Object, yr As String
    Dim fname As String, item As Object, numeroNode As Object
    Dim maxN As Long, parts() As String, n As Long
    fname = DataPath() & "ppap.xml"
    yr = Format$(Year(Date), "0000")
    If Dir(fname) <> "" Then
        Set dom = CreateObject("MSXML2.DOMDocument.6.0")
        dom.async = False
        If dom.Load(fname) Then
            Set root = dom.documentElement
            For Each item In root.ChildNodes
                Set numeroNode = item.selectSingleNode("Reference")
                If Not numeroNode Is Nothing Then
                    parts = Split(numeroNode.Text, "-")
                    If UBound(parts) >= 2 Then
                        If parts(1) = yr Then
                            n = CLng(Val(parts(2)))
                            If n > maxN Then maxN = n
                        End If
                    End If
                End If
            Next item
        End If
    End If
    m_Reference = "PPAP-" & yr & "-" & Format$(maxN + 1, "000")
End Sub

Public Function NbElementsFournis() As Long
    Dim i As Long, n As Long
    For i = 1 To 18: If m_El(i) Then n = n + 1: Next i
    NbElementsFournis = n
End Function

Public Function TauxConformite() As Double
    Dim requis As Long, i As Long
    Select Case m_NiveauPpap
        Case 1: requis = 1
        Case 2: requis = 3
        Case 3, 4, 5: requis = 18
    End Select
    If requis > 0 Then TauxConformite = NbElementsFournis() / requis
End Function

Public Function Validate() As Boolean
    If Len(m_ClientId) = 0 Then Exit Function
    If Len(Trim$(m_Piece)) = 0 Then Exit Function
    If m_NiveauPpap < 1 Or m_NiveauPpap > 5 Then Exit Function
    Validate = True
End Function
