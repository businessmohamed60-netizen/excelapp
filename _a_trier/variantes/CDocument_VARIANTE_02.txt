Attribute VB_Name = "CDocument"
Option Explicit

' ============================================================
' CDocument — Document du système qualité
' ISO 9001 §7.5 / IATF §7.5.3.2
' ============================================================

Private m_Id As String
Private m_Code As String               ' PR-QUA-001
Private m_Titre As String
Private m_TypeDoc As String            ' Procédure / Instruction / Formulaire / Enregistrement / Politique / Manuel
Private m_Version As String            ' 1.0, 2.1...
Private m_Statut As String             ' Brouillon / En revue / Approuvé / Obsolète
Private m_Redacteur As String
Private m_Approbateur As String
Private m_DateCreation As Date
Private m_DateApprobation As Date
Private m_DateRevuePrevue As Date
Private m_DateRevueRealisee As Date
Private m_Processus As String
Private m_CheminFichier As String      ' chemin relatif dans templates/docs
Private m_Commentaire As String

Public Property Get Id() As String: Id = m_Id: End Property
Public Property Let Id(ByVal v As String): m_Id = v: End Property
Public Property Get Code() As String: Code = m_Code: End Property
Public Property Let Code(ByVal v As String): m_Code = v: End Property
Public Property Get Titre() As String: Titre = m_Titre: End Property
Public Property Let Titre(ByVal v As String): m_Titre = v: End Property
Public Property Get TypeDoc() As String: TypeDoc = m_TypeDoc: End Property
Public Property Let TypeDoc(ByVal v As String): m_TypeDoc = v: End Property
Public Property Get Version() As String: Version = m_Version: End Property
Public Property Let Version(ByVal v As String): m_Version = v: End Property
Public Property Get Statut() As String: Statut = m_Statut: End Property
Public Property Let Statut(ByVal v As String): m_Statut = v: End Property
Public Property Get Redacteur() As String: Redacteur = m_Redacteur: End Property
Public Property Let Redacteur(ByVal v As String): m_Redacteur = v: End Property
Public Property Get Approbateur() As String: Approbateur = m_Approbateur: End Property
Public Property Let Approbateur(ByVal v As String): m_Approbateur = v: End Property
Public Property Get DateCreation() As Date: DateCreation = m_DateCreation: End Property
Public Property Let DateCreation(ByVal v As Date): m_DateCreation = v: End Property
Public Property Get DateApprobation() As Date: DateApprobation = m_DateApprobation: End Property
Public Property Let DateApprobation(ByVal v As Date): m_DateApprobation = v: End Property
Public Property Get DateRevuePrevue() As Date: DateRevuePrevue = m_DateRevuePrevue: End Property
Public Property Let DateRevuePrevue(ByVal v As Date): m_DateRevuePrevue = v: End Property
Public Property Get DateRevueRealisee() As Date: DateRevueRealisee = m_DateRevueRealisee: End Property
Public Property Let DateRevueRealisee(ByVal v As Date): m_DateRevueRealisee = v: End Property
Public Property Get Processus() As String: Processus = m_Processus: End Property
Public Property Let Processus(ByVal v As String): m_Processus = v: End Property
Public Property Get CheminFichier() As String: CheminFichier = m_CheminFichier: End Property
Public Property Let CheminFichier(ByVal v As String): m_CheminFichier = v: End Property
Public Property Get Commentaire() As String: Commentaire = m_Commentaire: End Property
Public Property Let Commentaire(ByVal v As String): m_Commentaire = v: End Property

Public Sub InitNew()
    m_Id = NewGuid()
    m_DateCreation = Now
    m_Statut = "Brouillon"
    m_Version = "1.0"
    m_TypeDoc = "Procédure"
    m_DateRevuePrevue = Date + 365
End Sub

Public Function RevueEchue() As Boolean
    RevueEchue = (m_Statut = "Approuvé") And (m_DateRevuePrevue < Date)
End Function

Public Function Validate() As Boolean
    If Len(Trim$(m_Code)) = 0 Then Validate = False: Exit Function
    If Len(Trim$(m_Titre)) = 0 Then Validate = False: Exit Function
    If m_Statut = "Approuvé" And Not IsDate(m_DateApprobation) Then
        Validate = False: Exit Function
    End If
    Validate = True
End Function

Public Function DisplayName() As String
    DisplayName = m_Code & " v" & m_Version & " — " & m_Titre
End Function
