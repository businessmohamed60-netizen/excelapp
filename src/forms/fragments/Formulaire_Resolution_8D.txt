Option Explicit
' MultiPage1 avec 8 pages D1..D8.
' Contrôles (préfixés) : txtD1_Champion, txtD1_Leader, txtD1_Membres,
'   txtD2_Description, txtD2_Date, txtD2_5W2H,
'   txtD3_Actions, txtD3_Date, txtD3_Verification,
'   txtD4_Ishikawa, txtD4_5Pourquoi, txtD4_CauseRacine, txtD4_CauseEchappee, txtD4_Date,
'   txtD5_Actions, txtD5_Responsable, txtD5_Date,
'   txtD6_Actions, txtD6_Date, txtD6_Validation,
'   txtD7_Actions, txtD7_Date, txtD8_Date, txtD8_Bilan,
'   lblNumero, lblEtape, cmdSave, cmdRapport, cmdClose

Private m_NcId As String, m_Current As C8D

Public Property Let NcCourante(ByVal id As String): m_NcId = id: End Property

Private Sub UserForm_Initialize()
    Caption = "Résolution 8D"
    cmdSave.Caption = T("btn.save"): cmdRapport.Caption = "Générer rapport 8D"
    cmdClose.Caption = T("btn.close")
    LoadOrCreate
End Sub

Private Sub LoadOrCreate()
    Dim huitds As Object, k As Variant
    Set huitds = g_CollMgr.LoadCollection("huitd")
    For Each k In huitds.Keys
        If huitds(k).NcId = m_NcId Then Set m_Current = huitds(k): Exit For
    Next k
    If m_Current Is Nothing Then
        Set m_Current = New C8D: m_Current.InitNew
        m_Current.NcId = m_NcId: m_Current.AssignNumero()
    End If
    LoadForm
End Sub

Private Sub LoadForm()
    lblNumero.Caption = "8D : " & m_Current.Numero & "  —  NC : " & m_NcId
    lblEtape.Caption = "Étape : " & m_Current.EtapeCourante()
    txtD1_Champion.Text = m_Current.D1_Champion: txtD1_Leader.Text = m_Current.D1_Leader
    txtD1_Membres.Text = m_Current.D1_Membres: txtD2_Description.Text = m_Current.D2_Description
    If m_Current.D2_Date > 0 Then txtD2_Date.Text = FormatIsoDate(m_Current.D2_Date)
    txtD2_5W2H.Text = m_Current.D2_5W2H: txtD3_Actions.Text = m_Current.D3_Actions
    If m_Current.D3_Date > 0 Then txtD3_Date.Text = FormatIsoDate(m_Current.D3_Date)
    txtD3_Verification.Text = m_Current.D3_Verification
    txtD4_Ishikawa.Text = m_Current.D4_Ishikawa: txtD4_5Pourquoi.Text = m_Current.D4_5Pourquoi
    txtD4_CauseRacine.Text = m_Current.D4_CauseRacine: txtD4_CauseEchappee.Text = m_Current.D4_CauseEchappee
    If m_Current.D4_Date > 0 Then txtD4_Date.Text = FormatIsoDate(m_Current.D4_Date)
    txtD5_Actions.Text = m_Current.D5_Actions: txtD5_Responsable.Text = m_Current.D5_Responsable
    If m_Current.D5_Date > 0 Then txtD5_Date.Text = FormatIsoDate(m_Current.D5_Date)
    txtD6_Actions.Text = m_Current.D6_Actions
    If m_Current.D6_Date > 0 Then txtD6_Date.Text = FormatIsoDate(m_Current.D6_Date)
    txtD6_Validation.Text = m_Current.D6_Validation: txtD7_Actions.Text = m_Current.D7_Actions
    If m_Current.D7_Date > 0 Then txtD7_Date.Text = FormatIsoDate(m_Current.D7_Date)
    If m_Current.D8_Date > 0 Then txtD8_Date.Text = FormatIsoDate(m_Current.D8_Date)
    txtD8_Bilan.Text = m_Current.D8_Bilan
End Sub

Private Sub SaveFromForm()
    m_Current.D1_Champion = txtD1_Champion.Text: m_Current.D1_Leader = txtD1_Leader.Text
    m_Current.D1_Membres = txtD1_Membres.Text: m_Current.D2_Description = txtD2_Description.Text
    If Len(txtD2_Date.Text) > 0 Then m_Current.D2_Date = ParseIsoDate(txtD2_Date.Text)
    m_Current.D2_5W2H = txtD2_5W2H.Text: m_Current.D3_Actions = txtD3_Actions.Text
    If Len(txtD3_Date.Text) > 0 Then m_Current.D3_Date = ParseIsoDate(txtD3_Date.Text)
    m_Current.D3_Verification = txtD3_Verification.Text: m_Current.D4_Ishikawa = txtD4_Ishikawa.Text
    m_Current.D4_5Pourquoi = txtD4_5Pourquoi.Text: m_Current.D4_CauseRacine = txtD4_CauseRacine.Text
    m_Current.D4_CauseEchappee = txtD4_CauseEchappee.Text
    If Len(txtD4_Date.Text) > 0 Then m_Current.D4_Date = ParseIsoDate(txtD4_Date.Text)
    m_Current.D5_Actions = txtD5_Actions.Text: m_Current.D5_Responsable = txtD5_Responsable.Text
    If Len(txtD5_Date.Text) > 0 Then m_Current.D5_Date = ParseIsoDate(txtD5_Date.Text)
    m_Current.D6_Actions = txtD6_Actions.Text
    If Len(txtD6_Date.Text) > 0 Then m_Current.D6_Date = ParseIsoDate(txtD6_Date.Text)
    m_Current.D6_Validation = txtD6_Validation.Text: m_Current.D7_Actions = txtD7_Actions.Text
    If Len(txtD7_Date.Text) > 0 Then m_Current.D7_Date = ParseIsoDate(txtD7_Date.Text)
    If Len(txtD8_Date.Text) > 0 Then m_Current.D8_Date = ParseIsoDate(txtD8_Date.Text)
    m_Current.D8_Bilan = txtD8_Bilan.Text
    If m_Current.D8_Date > 0 Then
        m_Current.Statut = "Clôturé"
    ElseIf m_Current.D4_Date > 0 Or m_Current.D5_Date > 0 Then
        m_Current.Statut = "En cours"
    Else
        m_Current.Statut = "Ouvert"
    End If
End Sub

Private Sub cmdSave_Click()
    On Error GoTo ErrHandler
    SaveFromForm
    If Not m_Current.Validate() Then ShowError "Validation", "D2 et D4 obligatoires.": Exit Sub
    g_CollMgr.Save m_Current, "huitd": g_CollMgr.InvalidateCache "huitd"
    lblEtape.Caption = "Étape : " & m_Current.EtapeCourante()
    ShowInfo "8D", "8D " & m_Current.Numero & " enregistré."
    Exit Sub
ErrHandler:
    LogEx "F8D", "cmdSave_Click", Err: ShowError "Sauvegarde", Err.Description
End Sub

Private Sub cmdRapport_Click()
    On Error GoTo ErrHandler
    SaveFromForm
    Dim path As String: path = GenererRapport8D(m_Current)
    If PromptYesNo("Rapport 8D", "Fichier : " & path & vbCrLf & "L'ouvrir ?") Then Workbooks.Open path
    Exit Sub
ErrHandler:
    LogEx "F8D", "cmdRapport_Click", Err: ShowError "Rapport", Err.Description
End Sub

Private Sub cmdClose_Click(): Unload Me: End Sub
